(function () {
  'use strict';

  // -------------------------------------------------------------------

    {
      key: 'empleabilidad',

      question: '¿Con quién cuentas hoy para acompañarte en este proceso?',
      options: [
        { label: 'Red de apoyo sólida', score: 1 },

        { label: 'Casi siempre solo/a', score: 3 },
        { label: 'Completamente solo/a', score: 4 }
      ],
      archetype: 'La Reinventora en Solitario',
      tag: 'diagnostico-comunidad',
      description: 'Estás cargando este proceso sin la red que mereces tener. Tu prioridad es rodearte de personas que entiendan lo que estás viviendo, para dejar de reinventarte a solas.'
    }
  ];

  // -------------------------------------------------------------------
  // 2. ESTADO
  // -------------------------------------------------------------------
  var root = document.getElementById('ny-quiz');
  var currentQuestion = 0;
  var scores = {};

  var screens = {
    questions: root.querySelector('[data-screen-id="questions"]'),
    capture: root.querySelector('[data-screen-id="capture"]'),
    result: root.querySelector('[data-screen-id="result"]')
  };

  var progressBar = document.getElementById('ny-progress-bar');
  var qCount = document.getElementById('ny-q-count');
  var qText = document.getElementById('ny-q-text');
  var qOptions = document.getElementById('ny-q-options');

  function showScreen(name) {
    Object.keys(screens).forEach(function (key) {
      screens[key].classList.toggle('ny-screen--active', key === name);
    });
    root.setAttribute('data-screen', name);
  }

  // -------------------------------------------------------------------
  // 3. ARRANQUE DEL QUIZ — llamado desde el botón nativo de Bricks.
  //    Oculta la sección de portada nativa y revela el quiz en su lugar.
  // -------------------------------------------------------------------
  window.nyStartQuiz = function () {
    // IMPORTANTE: ponle id="ny-cover-section" al CONTENEDOR COMPLETO de tu
    // portada nativa en Bricks (la sección/div que envuelve el eyebrow +
    // título + párrafo + botón) — no solo al botón. Así se oculta todo el
    // bloque de portada cuando arranca el quiz.
    var cover = document.getElementById('ny-cover-section');
    if (cover) cover.style.display = 'none';

    currentQuestion = 0;
    scores = {};
    root.classList.add('ny-quiz--visible');
    renderQuestion();
    showScreen('questions');
    root.scrollIntoView({ behavior: 'smooth', block: 'start' });
  };

  function renderQuestion() {
    var pillar = PILLARS[currentQuestion];
    qCount.textContent = 'Pregunta ' + (currentQuestion + 1) + ' de ' + PILLARS.length;
    qText.textContent = pillar.question;
    qOptions.innerHTML = '';

    pillar.options.forEach(function (opt) {
      var btn = document.createElement('button');
      btn.type = 'button';
      btn.className = 'ny-option';
      btn.textContent = opt.label;
      btn.addEventListener('click', function () {
        scores[pillar.key] = opt.score;
        advance();
      });
      qOptions.appendChild(btn);
    });

    progressBar.style.width = (currentQuestion / PILLARS.length * 100) + '%';
  }

  function advance() {
    currentQuestion++;
    if (currentQuestion < PILLARS.length) {
      renderQuestion();
    } else {
      progressBar.style.width = '100%';
      showScreen('capture');
    }
  }

  // -------------------------------------------------------------------
  // 4. CÁLCULO DEL ARQUETIPO GANADOR
  // -------------------------------------------------------------------
  function computeWinner() {
    var winner = PILLARS[0];
    var winnerScore = scores[PILLARS[0].key] || 0;

    for (var i = 1; i < PILLARS.length; i++) {
      var p = PILLARS[i];
      var s = scores[p.key] || 0;
      if (s > winnerScore) {
        winner = p;
        winnerScore = s;
      }
    }
    return winner;
  }

  // -------------------------------------------------------------------
  // 5. CAPTURA + ENVÍO AL ENDPOINT REST DE WORDPRESS
  // -------------------------------------------------------------------
  var form = document.getElementById('ny-capture-form');
  var submitBtn = document.getElementById('ny-submit-btn');
  var errorEl = document.getElementById('ny-form-error');

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    errorEl.textContent = '';

    var name = document.getElementById('ny-name').value.trim();
    var email = document.getElementById('ny-email').value.trim();
    var honeypot = document.getElementById('ny-hp').value.trim();

    if (honeypot) {
      showResultScreen(computeWinner());
      return;
    }

    if (!name) {
      errorEl.textContent = 'Por favor escribe tu nombre.';
      return;
    }
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      errorEl.textContent = 'Por favor escribe un correo válido.';
      return;
    }

    var winner = computeWinner();

    submitBtn.disabled = true;
    submitBtn.textContent = 'Enviando...';

    fetch('/wp-json/nextyou/v1/quiz-submit', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        name: name,
        email: email,
        scores: scores,
        archetype: winner.archetype,
        tag: winner.tag
      })
    })
      .then(function (res) {
        if (!res.ok) { throw new Error('Request failed'); }
        return res.json();
      })
      .then(function () {
        showResultScreen(winner);
      })
      .catch(function () {
        console.error('nextyou quiz-submit: no se pudo guardar el lead.');
        showResultScreen(winner);
      })
      .finally(function () {
        submitBtn.disabled = false;
        submitBtn.textContent = 'Ver mi resultado';
      });
  });

  function showResultScreen(winner) {
    document.getElementById('ny-result-archetype').textContent = winner.archetype;
    document.getElementById('ny-result-pillar').textContent = 'Pilar prioritario: ' + winner.key;
    document.getElementById('ny-result-description').textContent = winner.description;
    showScreen('result');
  }

  // -------------------------------------------------------------------
  // 6. CONEXIÓN CON EL BOTÓN NATIVO DE BRICKS
  //    Ponle id="ny-start-btn" a tu botón "Empezar diagnóstico" nativo
  //    (Settings del elemento en Bricks → campo "ID").
  // -------------------------------------------------------------------
  document.addEventListener('DOMContentLoaded', function () {
    var startBtn = document.getElementById('ny-start-btn');
    if (startBtn) {
      startBtn.addEventListener('click', function (e) {
        e.preventDefault();
        window.nyStartQuiz();
      });
    } else {
      console.warn('nextyou quiz: no se encontró #ny-start-btn. Revisa el ID del botón nativo en Bricks.');
    }
  });

})();
</script>
